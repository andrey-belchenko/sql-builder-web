  SELECT kod_sf.kodp AS kodp,                                       /*string*/
                                                                       /*key*/
         qube.m_facras_codeabon1 AS codeabon,                       /*number*/
                                                                       /*key*/
         kod_sf.kod_sf AS kod_sf,                                   /*number*/
                                                                       /*key*/
         kod_sf.kod_deb AS kod_deb,                                 /*number*/
                                                                       /*key*/
         MAX (kod_sf.name) AS sf_name,                              /*string*/
         kod_bu_akt.kod_bu_akt AS kod_bu_akt,                       /*number*/
                                                                       /*key*/
         MAX (kod_bu_akt.akt_name) AS akt_name,                     /*string*/
         SUM (qube.m_facras_summa1) AS nachisl,            /*Начислено (руб)*/
                                                                    /*number*/
         SUM (qube.m_opl_oplf1) AS opl,                             /*number*/
         SUM (qube.oborot1) AS dolg                        /*Начислено (руб)*/
                                                                    /*number*/
    FROM (  --
            SELECT ovr1.kod_sf AS kod_sf,                           /*number*/
                   ovr1.kod_bu_akt AS kod_bu_akt,                   /*number*/
                   MAX (ovr1.m_facras_codeabon1) AS m_facras_codeabon1, /*number*/
                   SUM (ovr1.m_facras_summa1) AS m_facras_summa1, /*Начислено (руб)*/
                                                                    /*number*/
                   SUM (ovr1.m_opl_oplf1) AS m_opl_oplf1,           /*number*/
                   SUM (ovr1.oborot1) AS oborot1,          /*Начислено (руб)*/
                                                                    /*number*/
                   SUM (ovr1.deb_dolg1) AS deb_dolg1       /*Начислено (руб)*/
                                                                    /*number*/
              FROM (--
                    SELECT qube.kod_sf AS kod_sf,                   /*number*/
                                                                       /*key*/
                           qube.kod_bu_akt AS kod_bu_akt,           /*number*/
                                                                       /*key*/
                           qube.m_facras_codeabon1 AS m_facras_codeabon1, /*number*/
                           qube.m_facras_summa1 AS m_facras_summa1, /*Начислено (руб)*/
                                                                    /*number*/
                           qube.m_opl_oplf1 AS m_opl_oplf1,         /*number*/
                             NVL (qube.m_facras_summa1, 0)
                           - NVL (qube.m_opl_oplf1, 0)
                               AS oborot1,                 /*Начислено (руб)*/
                                                                    /*number*/
                           SUM (
                               (  NVL (qube.m_facras_summa1, 0)
                                - NVL (qube.m_opl_oplf1, 0)))
                           OVER (PARTITION BY kod_sf.kod_deb)
                               AS deb_dolg1                /*Начислено (руб)*/
                                                                    /*number*/
                      FROM (--
                            SELECT qube.kod_sf AS kod_sf,           /*number*/
                                   qube.kod_bu_akt AS kod_bu_akt,   /*number*/
                                   qube.m_facras_codeabon1 AS m_facras_codeabon1, /*number*/
                                   qube.m_facras_summa1 AS m_facras_summa1, /*Начислено (руб)*/
                                                                            /*number*/
                                   qube.m_opl_oplf1 AS m_opl_oplf1  /*number*/
                              FROM (((--
                                      SELECT qube.kod_bu_akt AS kod_bu_akt, /*number*/
                                             qube.kod_sf AS kod_sf, /*number*/
                                             qube.m_facras_codeabon1
                                                 AS m_facras_codeabon1, /*number*/
                                             qube.m_facras_summa1
                                                 AS m_facras_summa1, /*Начислено (руб)*/
                                                                     /*number*/
                                             qube.m_opl_oplf1 AS m_opl_oplf1 /*number*/
                                        FROM (  --
                                                SELECT un.kod_bu_akt AS kod_bu_akt, /*number*/
                                                                                    /*key*/
                                                       un.kod_sf AS kod_sf, /*number*/
                                                                            /*key*/
                                                       MAX (un.m_facras_codeabon1)
                                                           AS m_facras_codeabon1, /*number*/
                                                       SUM (un.m_facras_summa1)
                                                           AS m_facras_summa1, /*Начислено (руб)*/
                                                                               /*number*/
                                                       SUM (un.m_opl_oplf1)
                                                           AS m_opl_oplf1 /*number*/
                                                  FROM (( (--
                                                           SELECT dims.kod_bu_akt
                                                                      AS kod_bu_akt, /*number*/
                                                                  dims.kod_sf
                                                                      AS kod_sf, /*number*/
                                                                  m_facras.codeabon
                                                                      AS m_facras_codeabon1, /*number*/
                                                                  m_facras.summa
                                                                      AS m_facras_summa1, /*Начислено (руб)*/
                                                                                          /*number*/
                                                                  NULL
                                                                      AS m_opl_oplf1 /*number*/
                                                             FROM (  --
                                                                     SELECT m_facras_a_d.kod_ras
                                                                                AS kod_ras_prm, /*number*/
                                                                                                /*key*/
                                                                            kod_bu_akt.kod_bu_akt
                                                                                AS kod_bu_akt, /*number*/
                                                                                               /*key*/
                                                                            m_facras_a_d.kod_sf
                                                                                AS kod_sf /*number*/
                                                                                          /*key*/
                                                                       FROM (--m_facras
                                                                             SELECT a.kod_ras
                                                                                        AS kod_ras, /*number*/
                                                                                                    /*key*/
                                                                                    a.kod_sf
                                                                                        AS kod_sf /*number*/
                                                                               FROM m_facras a--\m_facras
                                                                            )
                                                                            m_facras_a_d
                                                                            --\m_facras
                                                                            LEFT OUTER JOIN
                                                                            (--m_facvip
                                                                             SELECT a.kod_sf
                                                                                        AS kod_sf, /*number*/
                                                                                                   /*key*/
                                                                                    a.kodp
                                                                                        AS kodp, /**/
                                                                                                 /*string*/
                                                                                      (NVL (
                                                                                           kod_deb.dat_deb,
                                                                                           a.dat_sf))
                                                                                    + (1)
                                                                                        AS dat_bzad_uved /*date*/
                                                                               FROM m_facvip a
                                                                                    --\m_facvip
                                                                                    LEFT OUTER JOIN
                                                                                    (--m_debet
                                                                                     SELECT a.kod_deb
                                                                                                AS kod_deb, /*number*/
                                                                                                            /*key*/
                                                                                            a.dat_deb
                                                                                                AS dat_deb /**/
                                                                                                           /*date*/
                                                                                       FROM m_debet a--\m_debet
                                                                                    )
                                                                                    kod_deb
                                                                                        ON a.kod_deb =
                                                                                               kod_deb.kod_deb --\m_debet
                                                                                                              )
                                                                            kod_sf
                                                                                ON m_facras_a_d.kod_sf =
                                                                                       kod_sf.kod_sf --\m_facvip
                                                                            LEFT OUTER JOIN
                                                                            (--mv_recalc_akt
                                                                             SELECT a.kod_sf
                                                                                        AS kod_sf, /*number*/
                                                                                    a.kod_bu_akt
                                                                                        AS kod_bu_akt /*number*/
                                                                               FROM mv_recalc a
                                                                              --\mv_recalc
                                                                              WHERE a.kod_bu_akt
                                                                                        IS NOT NULL)
                                                                            mv_recalc_akt_a_d1
                                                                                ON mv_recalc_akt_a_d1.kod_sf =
                                                                                       kod_sf.kod_sf --\mv_recalc_akt
                                                                            LEFT OUTER JOIN
                                                                            (--hr_bu_akr
                                                                             SELECT a.kod_bu_akt
                                                                                        AS kod_bu_akt /*number*/
                                                                                                      /*key*/
                                                                               FROM hr_bu_akr a--\hr_bu_akr
                                                                            )
                                                                            kod_bu_akt
                                                                                ON mv_recalc_akt_a_d1.kod_bu_akt =
                                                                                       kod_bu_akt.kod_bu_akt --\hr_bu_akr
                                                                      WHERE     (kod_sf.kodp =
                                                                                     (NVL (
                                                                                          '98157',
                                                                                          kod_sf.kodp)))
                                                                            AND (kod_sf.dat_bzad_uved <
                                                                                     TO_DATE (
                                                                                         '07.09.2017',
                                                                                         'DD.MM.YYYY'))
                                                                   GROUP BY m_facras_a_d.kod_ras, /*number*/
                                                                                                  /*key*/
                                                                            kod_bu_akt.kod_bu_akt, /*number*/
                                                                                                   /*key*/
                                                                            m_facras_a_d.kod_sf /*number*/
                                                                                                /*key*/
                                                                                               )
                                                                  dims
                                                                  --\
                                                                  LEFT OUTER JOIN
                                                                  (--m_facras
                                                                   SELECT a.kod_ras
                                                                              AS kod_ras, /*number*/
                                                                                          /*key*/
                                                                          a.codeabon
                                                                              AS codeabon, /**/
                                                                                           /*number*/
                                                                            NVL (
                                                                                a.nachisl,
                                                                                0)
                                                                          + NVL (
                                                                                a.nds,
                                                                                0)
                                                                              AS summa /*Начислено (руб)*/
                                                                                       /*number*/
                                                                     FROM m_facras a--\m_facras
                                                                  ) m_facras
                                                                      ON m_facras.kod_ras =
                                                                             dims.kod_ras_prm --\m_facras
                                                                                             )
                                                         --\
                                                         UNION ALL
                                                         (--
                                                          SELECT dims.kod_bu_akt
                                                                     AS kod_bu_akt, /*number*/
                                                                 dims.kod_sf
                                                                     AS kod_sf, /*number*/
                                                                 NULL
                                                                     AS m_facras_codeabon1, /*number*/
                                                                 NULL
                                                                     AS m_facras_summa1, /*number*/
                                                                 m_opl.oplf
                                                                     AS m_opl_oplf1 /*number*/
                                                            FROM (  --
                                                                    SELECT m_opl_a_d.kod_opl
                                                                               AS kod_opl_prm, /*number*/
                                                                                               /*key*/
                                                                           kod_bu_akt.kod_bu_akt
                                                                               AS kod_bu_akt, /*number*/
                                                                                              /*key*/
                                                                           m_opl_a_d.kod_sf
                                                                               AS kod_sf /*number*/
                                                                                         /*key*/
                                                                      FROM (--m_opl
                                                                            SELECT a.kod_opl
                                                                                       AS kod_opl, /*number*/
                                                                                                   /*key*/
                                                                                   a.kod_sf
                                                                                       AS kod_sf, /*number*/
                                                                                   GREATEST (
                                                                                       a.datao,
                                                                                       kod_sf.dat_bzad_uved)
                                                                                       AS dat_dolg /*date*/
                                                                              FROM m_opl a
                                                                                   --\m_opl
                                                                                   LEFT OUTER JOIN
                                                                                   (--m_facvip
                                                                                    SELECT a.kod_sf
                                                                                               AS kod_sf, /*number*/
                                                                                                          /*key*/
                                                                                             (NVL (
                                                                                                  kod_deb.dat_deb,
                                                                                                  a.dat_sf))
                                                                                           + (1)
                                                                                               AS dat_bzad_uved /*date*/
                                                                                      FROM m_facvip a
                                                                                           --\m_facvip
                                                                                           LEFT OUTER JOIN
                                                                                           (--m_debet
                                                                                            SELECT a.kod_deb
                                                                                                       AS kod_deb, /*number*/
                                                                                                                   /*key*/
                                                                                                   a.dat_deb
                                                                                                       AS dat_deb /**/
                                                                                                                  /*date*/
                                                                                              FROM m_debet a--\m_debet
                                                                                           )
                                                                                           kod_deb
                                                                                               ON a.kod_deb =
                                                                                                      kod_deb.kod_deb --\m_debet
                                                                                                                     )
                                                                                   kod_sf
                                                                                       ON a.kod_sf =
                                                                                              kod_sf.kod_sf --\m_facvip
                                                                                                           )
                                                                           m_opl_a_d
                                                                           --\m_opl
                                                                           LEFT OUTER JOIN
                                                                           (--m_facvip
                                                                            SELECT a.kod_sf
                                                                                       AS kod_sf, /*number*/
                                                                                                  /*key*/
                                                                                   a.kodp
                                                                                       AS kodp /**/
                                                                                               /*string*/
                                                                              FROM m_facvip a--\m_facvip
                                                                           ) kod_sf
                                                                               ON m_opl_a_d.kod_sf =
                                                                                      kod_sf.kod_sf --\m_facvip
                                                                           LEFT OUTER JOIN
                                                                           (--mv_recalc_akt
                                                                            SELECT a.kod_sf
                                                                                       AS kod_sf, /*number*/
                                                                                   a.kod_bu_akt
                                                                                       AS kod_bu_akt /*number*/
                                                                              FROM mv_recalc a
                                                                             --\mv_recalc
                                                                             WHERE a.kod_bu_akt
                                                                                       IS NOT NULL)
                                                                           mv_recalc_akt_a_d1
                                                                               ON mv_recalc_akt_a_d1.kod_sf =
                                                                                      kod_sf.kod_sf --\mv_recalc_akt
                                                                           LEFT OUTER JOIN
                                                                           (--hr_bu_akr
                                                                            SELECT a.kod_bu_akt
                                                                                       AS kod_bu_akt /*number*/
                                                                                                     /*key*/
                                                                              FROM hr_bu_akr a--\hr_bu_akr
                                                                           )
                                                                           kod_bu_akt
                                                                               ON mv_recalc_akt_a_d1.kod_bu_akt =
                                                                                      kod_bu_akt.kod_bu_akt --\hr_bu_akr
                                                                     WHERE     (kod_sf.kodp =
                                                                                    (NVL (
                                                                                         '98157',
                                                                                         kod_sf.kodp)))
                                                                           AND (m_opl_a_d.dat_dolg <
                                                                                    TO_DATE (
                                                                                        '07.09.2017',
                                                                                        'DD.MM.YYYY'))
                                                                  GROUP BY m_opl_a_d.kod_opl, /*number*/
                                                                                              /*key*/
                                                                           kod_bu_akt.kod_bu_akt, /*number*/
                                                                                                  /*key*/
                                                                           m_opl_a_d.kod_sf /*number*/
                                                                                            /*key*/
                                                                                           )
                                                                 dims
                                                                 --\
                                                                 LEFT OUTER JOIN
                                                                 (--m_opl
                                                                  SELECT a.kod_opl
                                                                             AS kod_opl, /*number*/
                                                                                         /*key*/
                                                                           NVL (
                                                                               a.opl,
                                                                               0)
                                                                         + NVL (
                                                                               a.opls,
                                                                               0)
                                                                             AS oplf /*number*/
                                                                    FROM m_opl a--\m_opl
                                                                 ) m_opl
                                                                     ON m_opl.kod_opl =
                                                                            dims.kod_opl_prm --\m_opl
                                                                                            )--\
                                                       )--\
                                                       ) un
                                              --\
                                              GROUP BY un.kod_bu_akt, /*number*/
                                                                       /*key*/
                                                                     un.kod_sf /*number*/
                                                                               /*key*/
                                                                              )
                                             qube--\
                                   )--\
                                   )--\
                                   ) qube--\
                           ) qube
                           --\
                           LEFT OUTER JOIN
                           (--m_facvip
                            SELECT a.kod_sf AS kod_sf,              /*number*/
                                                                       /*key*/
                                                      a.kod_deb AS kod_deb /*number*/
                              FROM m_facvip a--\m_facvip
                           ) kod_sf
                               ON kod_sf.kod_sf = qube.kod_sf      --\m_facvip
                                                             ) ovr1
          --\
          GROUP BY ovr1.kod_sf,                                     /*number*/
                               ovr1.kod_bu_akt                      /*number*/
                                              ) qube
         --\
         LEFT OUTER JOIN
         (--hr_bu_akr
          SELECT a.kod_bu_akt AS kod_bu_akt,                        /*number*/
                                                                       /*key*/
                 '№ ' || a.num || ' от ' || (TO_CHAR (a.dat, 'DD.MM.YYYY'))
                     AS akt_name                                    /*string*/
            FROM hr_bu_akr a--\hr_bu_akr
         ) kod_bu_akt
             ON kod_bu_akt.kod_bu_akt = qube.kod_bu_akt           --\hr_bu_akr
         LEFT OUTER JOIN
         (--m_facvip
          SELECT a.kod_sf AS kod_sf,                                /*number*/
                                                                       /*key*/
                 a.kod_deb AS kod_deb,                              /*number*/
                 a.kodp AS kodp,                                          /**/
                                                                    /*string*/
                    '№ '
                 || a.num_sf
                 || ' от '
                 || (TO_CHAR (a.dat_sf, 'DD.MM.YYYY'))
                     AS name                                        /*string*/
            FROM m_facvip a--\m_facvip
         ) kod_sf
             ON kod_sf.kod_sf = qube.kod_sf                        --\m_facvip
   WHERE     (NVL (qube.oborot1, 0) != 0)
         AND (NVL (qube.deb_dolg1, 0) != 0)
         AND (kod_bu_akt.kod_bu_akt IS NOT NULL)
  HAVING (SUM (qube.oborot1) > 0)
GROUP BY kod_sf.kodp,                                               /*string*/
                                                                       /*key*/
         qube.m_facras_codeabon1,                                   /*number*/
                                                                       /*key*/
         kod_sf.kod_sf,                                             /*number*/
                                                                       /*key*/
         kod_sf.kod_deb,                                            /*number*/
                                                                       /*key*/
         kod_bu_akt.kod_bu_akt                                      /*number*/
                                                                       /*key*/
